    95	    fun show(service: AccessibilityService) {
    96	        if (composeView != null) return
    97	        
    98	        windowManager = service.getSystemService(Context.WINDOW_SERVICE) as WindowManager
    99	        
   100	        composeView = ComposeView(service).apply {
   101	            setContent {
   102	                var isVisible by remember { mutableStateOf(false) }
   103	                LaunchedEffect(Unit) {
   104	                    isVisible = true
   105	                }
   106	                
   107	                OmniscientAudioDockContent(
   108	                    service = service,
   109	                    isVisible = isVisible,
   110	                    onDismiss = {
   111	                        isVisible = false
   112	                        postDelayed({ dismiss(service) }, 250)
   113	                    }
   114	                )
   115	            }
   116	        }
   117	
   118	        lifecycleOwner = OverlayLifecycleOwner()
   119	        lifecycleOwner?.attachToView(composeView!!)
   120	        lifecycleOwner?.onCreate()
   121	        lifecycleOwner?.onStart()
   122	        lifecycleOwner?.onResume()
   123	
   124	        val params = WindowManager.LayoutParams(
   125	            WindowManager.LayoutParams.MATCH_PARENT,
   126	            WindowManager.LayoutParams.MATCH_PARENT,
   127	            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) WindowManager.LayoutParams.TYPE_ACCESSIBILITY_OVERLAY else WindowManager.LayoutParams.TYPE_PHONE,
   128	            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or 
   129	            WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL or 
   130	            WindowManager.LayoutParams.FLAG_WATCH_OUTSIDE_TOUCH or
   131	            WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
   132	            PixelFormat.TRANSLUCENT
   133	        ).apply {
   134	            gravity = Gravity.CENTER
   135	            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
   136	                flags = flags or WindowManager.LayoutParams.FLAG_BLUR_BEHIND
   137	                blurBehindRadius = 120
   138	            }
   139	        }
   140	
   141	        windowManager?.addView(composeView, params)
   142	
   143	        // Register Volume Receiver
   144	        val audioManager = service.getSystemService(Context.AUDIO_SERVICE) as AudioManager
   145	        val maxVol = audioManager.getStreamMaxVolume(AudioManager.STREAM_MUSIC).toFloat()
   146	        masterVolumeState.floatValue = audioManager.getStreamVolume(AudioManager.STREAM_MUSIC).toFloat()
   147	        
   148	        volumeReceiver = object : BroadcastReceiver() {
   149	            override fun onReceive(ctx: Context, intent: Intent) {
   150	                if (intent.action == "android.media.VOLUME_CHANGED_ACTION") {
   151	                    val streamType = intent.getIntExtra("android.media.EXTRA_VOLUME_STREAM_TYPE", -1)
   152	                    if (streamType == AudioManager.STREAM_MUSIC) {
   153	                        val currentVol = intent.getIntExtra("android.media.EXTRA_VOLUME_STREAM_VALUE", -1)
   154	                        if (currentVol != -1) {
   155	                            masterVolumeState.floatValue = currentVol.toFloat()
   156	                        }
   157	                    }
   158	                }
   159	            }
   160	        }
   161	        service.registerReceiver(volumeReceiver, IntentFilter("android.media.VOLUME_CHANGED_ACTION"))
   162	    }

   164	    fun dismiss(service: Context) {
   165	        volumeReceiver?.let {
   166	            try { service.unregisterReceiver(it) } catch (e: Exception) {}
   167	            volumeReceiver = null
   168	        }
   169	        composeView?.let {
   170	            lifecycleOwner?.onPause()
   171	            lifecycleOwner?.onStop()
   172	            lifecycleOwner?.onDestroy()
   173	            lifecycleOwner = null
   174	            windowManager?.removeView(it)
   175	            composeView = null
   176	        }
   177	        windowManager = null
   178	    }

   205	    private fun getRawActiveAudioPackages(context: Context): List<String> {
   206	        val audioManager = context.getSystemService(Context.AUDIO_SERVICE) as AudioManager
   207	        val pm = context.packageManager
   208	        val configs = audioManager.activePlaybackConfigurations
   209	        val pkgs = mutableListOf<String>()
   210	        
   211	        for (config in configs) {
   212	            try {
   213	                val getClientUidMethod = config.javaClass.getMethod("getClientUid")
   214	                val uid = getClientUidMethod.invoke(config) as Int
   215	                if (uid > 10000) { // Ignore system UIDs
   216	                    val packages = pm.getPackagesForUid(uid)
   217	                    if (packages != null) {
   218	                        pkgs.addAll(packages)
   219	                    }
   220	                }
   221	            } catch (e: Exception) {
   222	                // Ignore
   223	            }
   224	        }
   225	        return pkgs.distinct()
   226	    }
